cmake_minimum_required(VERSION 3.16)
project(ThreadForge VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

set(THREADFORGE_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE THREADFORGE_SANITIZER PROPERTY STRINGS none address thread)
if(NOT THREADFORGE_SANITIZER STREQUAL "none")
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer configuration requires GCC or Clang")
    endif()
    if(THREADFORGE_SANITIZER STREQUAL "address")
        set(SANITIZER_FLAG -fsanitize=address,undefined)
    elseif(THREADFORGE_SANITIZER STREQUAL "thread")
        set(SANITIZER_FLAG -fsanitize=thread)
    else()
        message(FATAL_ERROR "Unknown THREADFORGE_SANITIZER")
    endif()
    add_compile_options(${SANITIZER_FLAG} -fno-omit-frame-pointer)
    add_link_options(${SANITIZER_FLAG})
endif()

add_library(threadforge src/thread_pool.cpp src/scheduler.cpp src/metrics.cpp)
target_include_directories(threadforge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(threadforge PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic)
    target_compile_options(threadforge PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(basic_usage examples/basic_usage.cpp)
target_link_libraries(basic_usage PRIVATE threadforge)
add_executable(parallel_tasks examples/parallel_tasks.cpp)
target_link_libraries(parallel_tasks PRIVATE threadforge)
add_executable(priority_demo examples/priority_demo.cpp)
target_link_libraries(priority_demo PRIVATE threadforge)
add_executable(threadforge_benchmark benchmarks/benchmark_main.cpp benchmarks/workloads.cpp)
target_link_libraries(threadforge_benchmark PRIVATE threadforge)

include(CTest)
if(BUILD_TESTING)
    add_executable(thread_pool_tests tests/thread_pool_tests.cpp)
    target_link_libraries(thread_pool_tests PRIVATE threadforge)
    add_test(NAME thread_pool_tests COMMAND thread_pool_tests)
    add_executable(stress_tests tests/stress_tests.cpp)
    target_link_libraries(stress_tests PRIVATE threadforge)
    add_test(NAME stress_tests COMMAND stress_tests)
    set_tests_properties(stress_tests PROPERTIES TIMEOUT 180 LABELS stress)
    add_test(NAME basic_usage COMMAND basic_usage)
    add_test(NAME parallel_tasks COMMAND parallel_tasks)
    add_test(NAME priority_demo COMMAND priority_demo)
    add_test(NAME benchmark_smoke COMMAND threadforge_benchmark --tasks 5 --repeats 1 --iterations 20 --max-workers 2)
    set_tests_properties(thread_pool_tests basic_usage parallel_tasks priority_demo benchmark_smoke PROPERTIES TIMEOUT 60)
endif()
